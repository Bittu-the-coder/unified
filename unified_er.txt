Table users {
  id string [pk, note: 'UUID']
  email string [unique, not null]
  phone string [unique]
  username string [unique, not null]
  full_name varchar(200)
  password_hash varchar(255) [not null, note: 'BCrypt hashed']
  salt varchar(255) [not null]
  avatar_url text
  bio text
  role varchar(50) [not null, default: 'user']
  status varchar(50) [not null, default: 'active']
  email_verified boolean [default: false]
  phone_verified boolean [default: false]
  two_factor_enabled boolean [default: false]
  two_factor_secret varchar(255)
  last_login timestamp
  last_active timestamp
  login_streak integer [default: 0]
  preferences json [note: 'User preferences as JSON']
  created_at timestamp [default: `now()`, not null]
  updated_at timestamp [default: `now()`, not null]
  deleted_at timestamp

  indexes {
    (email) [unique]
    (phone) [unique]
    (username) [unique]
    (status)
    (role, status)
  }
}

Table ai_conversations {
  id string [pk, note: 'UUID']
  user_id string [not null]
  title varchar(500)
  context json
  created_at timestamp [default: `now()`, not null]
  updated_at timestamp [default: `now()`, not null]

  indexes {
    (user_id, updated_at)
    (user_id, created_at)
  }
}

Table ai_messages {
  id string [pk, note: 'UUID']
  conversation_id string [not null]
  role varchar(50) [not null]
  content text [not null]
  tokens_used integer
  model_used varchar(200)
  metadata json
  created_at timestamp [default: `now()`, not null]

  indexes {
    (conversation_id, created_at)
    (conversation_id, role)
  }
}

Table ai_memory {
  id string [pk, note: 'UUID']
  user_id string [not null]
  memory_key varchar(500) [not null]
  memory_value json [not null]
  context varchar(1000)
  expires_at timestamp
  created_at timestamp [default: `now()`, not null]
  updated_at timestamp [default: `now()`, not null]

  indexes {
    (user_id, memory_key) [unique]
    (user_id, expires_at)
    (user_id, context)
  }
}

Table ai_suggestions {
  id string [pk, note: 'UUID']
  user_id string [not null]
  suggestion_type varchar(100) [not null]
  content json [not null]
  context_id string
  is_accepted boolean [default: false]
  is_dismissed boolean [default: false]
  created_at timestamp [default: `now()`, not null]

  indexes {
    (user_id, created_at)
    (user_id, suggestion_type, is_accepted)
    (user_id, is_dismissed)
  }
}

// ======================================================
// NOTIFICATIONS
// ======================================================

Table notifications {
  id string [pk, note: 'UUID']
  user_id string [not null]
  type varchar(100) [not null]
  title varchar(500) [not null]
  body text
  data json
  is_read boolean [default: false]
  is_clicked boolean [default: false]
  action_url text
  image_url text
  created_at timestamp [default: `now()`, not null]
  read_at timestamp

  indexes {
    (user_id, created_at)
    (user_id, is_read, created_at)
    (user_id, type, created_at)
    (user_id, is_clicked)
  }
}

Table notification_settings {
  id string [pk, note: 'UUID']
  user_id string [unique, not null]
  email_notifications json [default: '{}']
  push_notifications json [default: '{}']
  in_app_notifications json [default: '{}']
  quiet_hours_start varchar(10) [note: 'HH:MM format']
  quiet_hours_end varchar(10) [note: 'HH:MM format']
  quiet_hours_timezone varchar(100)
  updated_at timestamp [default: `now()`, not null]

  indexes {
    (user_id) [unique]
  }
}

// ======================================================
// ANALYTICS & USER ACTIVITY
// ======================================================

Table user_activities {
  id string [pk, note: 'UUID']
  user_id string [not null]
  activity_type varchar(200) [not null]
  entity_type varchar(200)
  entity_id string
  metadata json
  ip_address varchar(50)
  user_agent text
  created_at timestamp [default: `now()`, not null]

  indexes {
    (user_id, created_at)
    (activity_type, created_at)
    (entity_type, entity_id)
    (created_at)
  }
}

Table feature_usage {
  id string [pk, note: 'UUID']
  user_id string [not null]
  feature_name varchar(200) [not null]
  action varchar(200) [not null]
  duration integer [note: 'In seconds']
  metadata json
  created_at timestamp [default: `now()`, not null]

  indexes {
    (user_id, feature_name, created_at)
    (feature_name, action, created_at)
    (created_at)
  }
}

// ======================================================
// CODE PLAYGROUND (Optional)
// ======================================================

Table code_snippets {
  id string [pk, note: 'UUID']
  user_id string [not null]
  title varchar(500) [not null]
  description text
  language varchar(100) [not null]
  code text [not null]
  is_public boolean [default: false]
  tags json
  created_at timestamp [default: `now()`, not null]
  updated_at timestamp [default: `now()`, not null]

  indexes {
    (user_id, is_public)
    (user_id, language)
    (is_public, language)
    (created_at)
  }
}

Table code_executions {
  id string [pk, note: 'UUID']
  user_id string [not null]
  snippet_id string
  language varchar(100) [not null]
  code text [not null]
  input text
  output text
  error text
  execution_time integer [note: 'In milliseconds']
  memory_used integer [note: 'In KB']
  created_at timestamp [default: `now()`, not null]

  indexes {
    (user_id, created_at)
    (snippet_id)
    (language, created_at)
  }
}

Table coding_challenges {
  id string [pk, note: 'UUID']
  title varchar(500) [not null]
  description text [not null]
  difficulty varchar(50) [not null]
  category varchar(200)
  test_cases json [not null]
  solution_template json
  hints json
  created_at timestamp [default: `now()`, not null]
  updated_at timestamp [default: `now()`, not null]

  indexes {
    (difficulty, category)
    (category, created_at)
    (difficulty)
  }
}

Table challenge_submissions {
  id string [pk, note: 'UUID']
  user_id string [not null]
  challenge_id string [not null]
  code text [not null]
  language varchar(100) [not null]
  status varchar(50) [not null]
  execution_time integer
  memory_used integer
  test_cases_passed integer
  total_test_cases integer
  created_at timestamp [default: `now()`, not null]

  indexes {
    (user_id, challenge_id, created_at)
    (challenge_id, status)
    (user_id, status, created_at)
    (challenge_id, language)
  }
}
